# This first C target is the internal CPU linear kernel, not the engine API.
# A common C source selects small target primitives where measurements justify
# them. Clang lowers C through LLVM at build time; neither LLVM nor Rust is
# needed at runtime.
if(NOT CMAKE_C_COMPILER_ID MATCHES "Clang|GNU")
    message(FATAL_ERROR "The initial kernel build supports Clang and GCC")
endif()

# Require a compiler that recognizes the published standard's flag. CMake may
# use its equivalent -std=c2x spelling. GCC 14 still reports the draft macro
# value even in -std=c23 mode, so __STDC_VERSION__ cannot enforce this minimum.
include(CheckCCompilerFlag)
check_c_compiler_flag("-std=c23" VELRI_HAS_C23_FLAG)
if(NOT VELRI_HAS_C23_FLAG)
    message(FATAL_ERROR "Velri requires C23 mode; use Clang 18+ or GCC 14+")
endif()

function(velri_strict_target target)
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -fno-fast-math)
    if(VELRI_SANITIZE)
        target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
        target_link_options(${target} PRIVATE -fsanitize=address,undefined)
    endif()
endfunction()

# Generate narrower forms as well: small M must not allocate a full wide tile.
# The manifest serves both declarations and selection, so these cannot drift.
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/linear_variants.inc" "/* Generated by CMake. */\n")
function(velri_kernel_family name variant maximum pair_maximum)
    set(columns 1)
    while(columns LESS_EQUAL maximum)
        set(target velri_kernel_${name}_${columns})
        set(symbol velri_linear_${name}_c${columns})
        set(pair_columns ${columns})
        if(pair_columns GREATER pair_maximum)
            set(pair_columns ${pair_maximum})
        endif()
        add_library(${target} OBJECT src/backend/cpu/linear_kernel.c)
        target_compile_definitions(${target} PRIVATE VELRI_KERNEL=${symbol} VELRI_COLUMNS=${columns}
            VELRI_PAIR_COLUMNS=${pair_columns})
        velri_strict_target(${target})
        target_compile_options(${target} PRIVATE ${VELRI_BASE_FLAGS} ${ARGN} -ffp-contract=on)
        set_property(TARGET ${target} PROPERTY POSITION_INDEPENDENT_CODE ON)
        target_sources(velri_cpu_linear PRIVATE $<TARGET_OBJECTS:${target}>)
        file(APPEND "${CMAKE_CURRENT_BINARY_DIR}/linear_variants.inc"
            "VELRI_ENTRY(${symbol}, ${variant}, ${columns})\n")
        math(EXPR columns "${columns} * 2")
    endwhile()
endfunction()

velri_kernel_family(base VELRI_CPU_BASE ${VELRI_BASE_COLUMNS} ${VELRI_BASE_COLUMNS})
